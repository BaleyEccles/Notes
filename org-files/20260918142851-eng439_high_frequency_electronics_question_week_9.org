* Question 1
\begin{align*}
G_1 &= 10^{12/10} = 15.85 \\
F_1 &= 10^{4/10} = 2.512 \\
F_2 &= 1 + \frac{T_{e2}}{T_0} \\
F_2 &= 1 + \frac{900}{290} \\
F_2 &= 4.103 \\
F_{\textrm{total}} &= F_1 + \frac{F_2 - 1}{G_1} \\
F_{\textrm{total}} &= 2.512 + \frac{4.103 - 1}{15.85} \\
F_{\textrm{total}} &= 2.708 \\
NF_{\textrm{total}} &= 10\log_{10}(2.708) \\
NF_{\textrm{total}} &= 4.33\ \textrm{dB}
\end{align*}
#+BEGIN_SRC octave :results output :exports none :session Q1 :eval no-export :tangle ENG439_Week_9_Q1.m

F1_dB = 4;
Te2 = 900;
T0 = 290;

F1 = 10^(F1_dB/10)

F2 = 1 + Te2/T0

F_total = F1 + (F2 - 1)/G1

NF_total_dB = 10*log10(F_total)
#+END_SRC

* Question 2
\begin{align*}
F &= 1 + \frac{T_e}{T_0} \\
F &= 1 + \frac{250}{290} \\
F &= 1.862 \\
NF &= 10\log_{10}(1.862) \\
NF &= 2.70\ \textrm{dB} \\
\\
N &= kT_0B \\
N &= 1.38\times10^{-23}\cdot290\cdot10^9 \\
N &= 4.00\times10^{-12}\ \textrm{W} \\
N_{\textrm{dBm}} &=
10\log_{10}\left(\frac{4.00\times10^{-12}}{10^{-3}}\right) \\
N_{\textrm{dBm}} &= -84.0\ \textrm{dBm} \\
\\
N_o &= N_{\textrm{dBm}} + NF + G \\
N_o &= -84.0 + 2.70 + 15 \\
N_o &= -66.3\ \textrm{dBm} \\
\\
LDR &= OP_{1\textrm{dB}} - N_o \\
LDR &= 5 - (-66.3) \\
LDR &= 71.3\ \textrm{dB}
\end{align*}
#+BEGIN_SRC octave :results output :exports none :session Q2 :eval no-export :tangle ENG439_Week_9_Q2.m
T0 = 290;
B = 1e9;
G_dB = 15;
Te = 250;
P1dB_out = 5;
k = 1.38e-23;

F = 1 + Te/T0
NF_dB = 10*log10(F)

N = k*T0*B
N_dBm = 10*log10(N/1e-3)

No_dBm = N_dBm + NF_dB + G_dB

LDR = P1dB_out - No_dBm
#+END_SRC
