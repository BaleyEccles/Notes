* ENG439 - High Frequency Electronics - Formula Sheet
** Transmission Lines
\[Z_{in} = Z_0\frac{Z_L + j Z_0\tan(\beta l)}{Z_0 + j Z_L\tan(\beta l )}\]
\[\beta = \frac{2\pi}{\lambda}\]
\[\Gamma = \frac{Z_L - Z_0}{Z_L + Z_0}\]
\[VSWR = \frac{1 + |\Gamma|}{1 - |\Gamma|}\]

** Waveforms
\[V_g(t) = V_g\cos(\omega t)\]
\[\tilde{V}_{in} = \tilde{V}_g \frac{Z_{in}}{Z_g + Z_{in}}\]
\[\tilde{I}_{in} = \frac{\tilde{V}_in}{Z_{in}}\]
\[\tilde{V}_{in} = |V_{in}|\angle \phi\]
\[V_{in}(t) = |V_{in}|\cos(\omega t + \phi)\]

** Standing Waves
\[\Gamma = |\Gamma|e^{j\theta_r}\]
\[\Gamma(d) = \Gamma e^{-j2\beta d}\]
\[\theta_{\Gamma} - 2\beta d_{\max} = 2n\pi\]
\[\theta_{\Gamma} - 2\beta d_{\min} = (2n + 1)\pi\]
\[d_{\max} \rightarrow d_{\min} = \frac{\lambda}{4}\]
\[d_{\max_1} \rightarrow d_{\max_2} = \frac{\lambda}{2}\]
\[d_{\min_1} \rightarrow d_{\min_2} = \frac{\lambda}{2}\]

** TEM Transmission-Line Parameters
\[\gamma = \alpha + j\beta = \sqrt{(R' + j\omega L')(G' + j\omega C')}\]
\[Z_0 = \sqrt{\frac{R' + j\omega L'}{G' + j\omega C'}}\]
\[u_p = \frac{\omega}{\beta}\]
\[\beta_{TEM} \approx \omega\sqrt{L'C'}\]
\[Z_0_{TEM} \approx \sqrt{\frac{L'}{C'}}\]
\[u_p_{TEM} \approx \frac{1}{\sqrt{L'C'}}\]
\[u_p_{\mathrm{dialetric}} = \frac{3\times10^8}{\epsilon_r}\]

** Microstrip
\[\epsilon_{eff} \approx \frac{\epsilon_r + 1}{2} + \frac{\epsilon_r - 1}{2}\frac{1}{\sqrt{1 + 12h/W}}\]
\[Z_0 =
\begin{cases}
\dfrac{120\pi}
{\sqrt{\epsilon_{eff}}\left(1.393+\dfrac{W}{d}
+0.667\ln\left(\dfrac{W}{d}+1.444\right)\right)},
& \dfrac{W}{d}\ge 1, \\[1.2em]
\dfrac{60}{\sqrt{\epsilon_{eff}}}
\ln\left(\dfrac{8d}{W}+\dfrac{W}{4d}\right),
& \dfrac{W}{d}\le 1.
\end{cases}\]

\[\lambda_g = \frac{c}{f\sqrt{\epsilon_{eff}}}\]
\[l = \frac{\theta}{360^{\circ}}\lambda_g\]

** Rectangular Waveguides
\[f_{c,mn} = \frac{c}{2\sqrt{\epsilon_r}}\sqrt{\left(\frac{m}{a}\right)^2 + \left(\frac{n}{b}\right)^2}\]
\[m_{TE} + n_{TE} \neq 0\]
\[m_{TM}\cdot n_{TM} \neq 0\]

\[\mathrm{prpagate} = f > f_c\]
\[\beta = k\sqrt{1 - \left(\frac{f_c}{f}\right)^2}\]
\[k = \frac{2\pi}{\lambda}\]
\[\lambda_g = \frac{\lambda}{\sqrt{1 - \left(\frac{f_c}{f}\right)^2}\]
\[v_p = \frac{\frac{c}{\sqrt{\epsilon_r}}}{\sqrt{1 - \left(\frac{f_c}{f}\right)^2}\]
\[v_g = \frac{c}{\sqrt{\epsilon_r}}\sqrt{1 - \left(\frac{f_c}{f}\right)^2}\]
\[v_pv_g = \frac{c^2}{\epsilon_r}\]

** Z Parameters
\[\begin{bmatrix}
V_1 \\
V_2 \\
\vdots \\
V_N
\end{bmatrix} = \begin{bmatrix}
Z_{11} & Z_{12} & \hdots & Z_{1N} \\
Z_{21} & Z_{22} & \hdots & \vdots\\
\vdots& \hdots& \vdots& \vdots \\
Z_{N1} & \hdots & \hdots & Z_{NN}
\end{bmatrix}\begin{bmatrix}
I_1 \\
I_2 \\
\vdots \\
I_N
\end{bmatrix}\]
\[Z_{ij} = \frac{V_i}{I_j}|_{I_k = 0\textrm{ for } k\neq j}\]

** ABCD Parameters

\end{bmatrix} = \begin{bmatrix}

\end{bmatrix}\begin{bmatrix}

\[T_{A} &= \begin{bmatrix}
A_A & B_A \\

\end{bmatrix}\]
\[T_{B} &= \begin{bmatrix}
A_B & B_B \\

\[T_{\mathrm{total}} = T_AT_B\]

** S Parameters
\[\begin{bmatrix}
V^-_1 \\
V^-_2 \\
\vdots \\
V^-_N
\end{bmatrix} = \begin{bmatrix}
S_{11} & S_{12} & \hdots & S_{1N} \\
S_{21} & S_{22} & \hdots & \vdots\\
\vdots& \hdots& \vdots& \vdots \\
S_{N1} & \hdots & \hdots & S_{NN}
\end{bmatrix}\begin{bmatrix}
V^+_1 \\
V^+_2 \\
\vdots \\
V^+_N
\end{bmatrix}\]
\[S_{ij} = \frac{V^-_i}{V^+_j}|_{V^+_k = 0\textrm{ for } k\neq j}\]
\[S_{\textrm{matched}} = \begin{bmatrix}
0 & S_{12} & \hdots & S_{1N} \\
S_{21} & 0 & \hdots & \vdots\\
\vdots& \hdots& \vdots& \vdots \\
S_{N1} & \hdots & \hdots & 0
\end{bmatrix}\]
\[S_{\textrm{recip}} = S_{\textrm{recip}}^T\]
\[\sum_i|S_{ij,\textrm{lossless}}|^2 = 1\]
\[\sum|a_{i,\textrm{lossless}}|^2 = \sum|b_{i,\textrm{lossless}}|^2\]


** ABCD To S Conversion
\[\begin{aligned}
S_{11} &= \frac{-A+B/Z_0-CZ_0+D}{A+B/Z_0+CZ_0+D} \\
S_{12} &= \frac{2(AD-BC)}{A+B/Z_0+CZ_0+D} \\
S_{21} &= \frac{2}{A+B/Z_0+CZ_0+D} \\
S_{22} &=\frac{-A+B/Z_0+CZ_0+D}{A+B/Z_0+CZ_0+D}
\end{aligned}\]

** Even/Odd Mode Analysis
\[\Gamma = \frac{\Gamma_e + \Gamma_o}{2}\]
\[T = \frac{\Gamma_e - \Gamma_o}{2}\]

** Filters
\[L_k = \frac{Z_0g_k}{\omega_c\]
\[C_k = \frac{g_k}{Z_0\omega_c}\]

** Transformations
\[\Delta = \frac{\omega_2 - \omega_1}{\omega_0}\]
\[\omega_0 = \sqrt{\omega_1\omega_2}\]
*** High Pass
\[L_{LP} \rightarrow C_{HP} = \frac{1}{Z_0\omega_c g_k}\]
\[C_{LP} \rightarrow L_{HP} = \frac{Z_0}{\omega_c g_k}\]
*** Band Pass
\[L_{LP} \rightarrow L_{BP} &= \frac{g_kZ_0}{\omega_0 \Delta} +  C_{BP} &= \frac{\Delta}{Z_0\omega_0 g_k}\]
\[C_{LP} \rightarrow L_{BP} &= \frac{\Delta Z_0}{\omega_0 g_k} || C_{BP} &= \frac{g_k}{Z_0\omega_0 \Delta}\]
*** Band Stop
\[L_{LP} \rightarrow L_{BS} &= \frac{Z_0g_k\Delta}{\omega_0} ||   C_{BS} &= \frac{1}{Z_0\omega_0 g_k \Delta}\]
\[C_{LP} \rightarrow L_{BS} &= \frac{Z_0}{\omega_0 g_k \Delta} + C_{BS} &= \frac{g_k \Delta}{Z_0\omega_0}\]

** Richards' Transformation
\[L_{LP} \rightarrow Z_{\textrm{series stub}} = Z_0 g_k\]
\[L_{LP} \rightarrow Z_{\textrm{shunt stub}} = \frac{Z_0}{g_k}\]
** Kuroda's identities
\[n^2 = 1 + \frac{Z_{\textrm{stub}}}{Z_{\textrm{UE}}}\]
\[Z_{\textrm{UE}_1} = Z_0\]
\[Z_{\textrm{series stub}} \rightarrow Z_{\textrm{shunt stub}} = \begin{cases}
Z_{\textrm{UE}}' &= n^2 Z_{\textrm{stub}} \\
Z_{\textrm{stub}}' &= n^2 Z_{\textrm{UE}} \\
\end{cases}\]
\[Z_{\textrm{shunt stub}} \rightarrow Z_{\textrm{series stub}} = \begin{cases}
Z_{\textrm{UE}}' &= \frac{Z_{\textrm{stub}}}{n^2} \\
Z_{\textrm{stub}}' &= \frac{Z_{\textrm{UE}}}{n^2} \\
\end{cases}\]
